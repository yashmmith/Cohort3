8. Object of Objects
Create a function that takes an array of objects as input, and returns the users whose age > 18 and are male
let users=[{
    name:"Yash",
    age:18,
    gender:"Male"
},{
    name:"Jatin",
    age:20,
    gender:"Female"
}];
let filter=(users)=>{
    let na=users.filter((val)=>(val.age>=18 && val.gender==="Male"));
    console.log(na);
}
filter(users);
